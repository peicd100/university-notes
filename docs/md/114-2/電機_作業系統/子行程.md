```c
#include <sys/wait.h>
#include <sys/types.h>
#include <stdio.h>
#include <unistd.h>
#include <stdlib.h>

int main()
{
    pid_t pid;
    /* fork another process */
    pid = fork();
    if (pid < 0) { /* error occurred */
        fprintf(stderr, "Fork Failed");
        exit(-1);
    }
    else if (pid == 0) { /* child process */
        execlp("/bin/ls", "ls", NULL);
    }
    else { /* parent process */
        wait(NULL);
        printf("Child Complete");
        exit(0);
    }
}
```

```c
#include <stdio.h>
#include <unistd.h>
#include <sys/wait.h>

int main() {
    int x = 10;

    pid_t pid = fork();

    if (pid == 0) {
        x += 5;
        printf("Child: x = %d\n", x);
    } else {
        wait(NULL);
        x += 20;
        printf("Parent: x = %d\n", x);
    }

    return 0;
}
```
